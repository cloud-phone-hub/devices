## 6) 官方 rc 与 VINTF 声明（固件原件，dump.erofs）
$ cat odm_a/etc/init/android.hardware.power.stats-service.oplus.rc
service vendor.power.stats /odm/bin/hw/android.hardware.power.stats-impl.oplus
    class hal
    user system
    group system root
# 无 interface aidl 行 → 懒启动只会命中我们的服务；它靠 class_start hal 起
# （boot 阶段 10.606 s：Command 'class_start hal' action=boot 完成；early-init 在 0.81 s）

$ cat odm_a/etc/vintf/manifest/android.hardware.power.stats-service.oplus.xml
<manifest version="1.0" type="device">
    <hal format="aidl"><name>vendor.oplus.hardware.handlefactory</name><fqname>IHandleFactory/default</fqname></hal>
</manifest>
# ↑ 这份已被覆盖件顶成空 manifest（云机无实现、无客户端），与本条无关
